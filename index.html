Supabase SQL Schema Helper

Copy and run this SQL script in your Supabase SQL Editor to create the required inventory table with Row Level Security (RLS) configured for House of Anna: 

-- Create House of Anna inventory table
create table if not exists public.inventory (
  id bigint generated by default as identity primary key,
  product_id text unique not null,
  print_description text not null,
  supplier text not null,
  category text not null,
  size text not null,
  cost_price numeric not null default 0,
  selling_price numeric not null default 0,
  margin numeric not null default 0,
  total_qty_stocked integer not null default 1,
  arrival_date date not null default current_date,
  shelf_location text,
  photo_url text,
  status text not null default 'Available',
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- Enable Row Level Security (RLS)
alter table public.inventory enable row level security;

-- Policy to allow read access for all
create policy "Allow public read access"
on public.inventory for select
using (true);

-- Policy to allow insert access for anon users
create policy "Allow public insert access"
on public.inventory for insert
with check (true);

-- Policy to allow update status
create policy "Allow public update access"
on public.inventory for update
using (true);
